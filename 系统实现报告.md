## 一、系统架构

### 1.1 总体设计

系统采用**算法与可视化分离**的架构：算法层为纯函数（输入参数 → 状态快照数组），可视化层仅负责按时间顺序"播放"快照，不参与算法逻辑。

```
┌──────────────────────────────────────────────────┐
│           视图层 (Vue 3 组件)                     │
│  CpuSchedule.vue  Banker.vue  PageReplace.vue    │
│  GanttChart.vue   MemoryGrid.vue  BufferVisual.vue│
├──────────────────────────────────────────────────┤
│          可视化引擎 (ECharts 6)                    │
├──────────────────────────────────────────────────┤
│          算法层 (纯 JS 函数)                       │
│  cpuSchedule.js  banker.js  pageReplace.js        │
│  diskSchedule.js  producerConsumer.js            │
└──────────────────────────────────────────────────┘
```

### 1.2 数据流

```
用户配置参数 → 调用算法函数 → 得到状态快照数组 states[]
→ PlaybackControl 控制帧索引 → 渲染组件根据当前帧绘制图表
```

### 2.1 FCFS（先来先服务）

**算法思想**：按进程到达时间排序，依次执行，非抢占。

**伪代码**：

```
算法 FCFS(processes)
输入: processes — 进程列表 { id, arrivalTime, burstTime }
输出: { gantt[], results[] }

1.  sorted ← processes 按 arrivalTime 升序排列
2.  currentTime ← 0
3.  gantt ← [], results ← []
4.
5.  for each p ∈ sorted do
6.      if currentTime < p.arrivalTime then
7.          currentTime ← p.arrivalTime        // CPU 空闲，快进到进程到达
8.      end if
9.      start ← currentTime
10.     end ← start + p.burstTime
11.     gantt.push({ pid: p.id, start, end })  // 一次性执行完毕
12.     currentTime ← end
13.
14.     turnaroundTime ← end - p.arrivalTime
15.     weightedTurnaroundTime ← turnaroundTime / p.burstTime
16.     results.push({ id: p.id, finishTime: end, turnaroundTime, weightedTurnaroundTime })
17. end for
18.
19. return { gantt, results }
```

**关键代码**：

```js
export function fcfs(processes) {
  const sorted = [...processes].sort((a, b) => a.arrivalTime - b.arrivalTime)
  const gantt = []
  const results = []
  let currentTime = 0

  for (const p of sorted) {
    if (currentTime < p.arrivalTime) {
      currentTime = p.arrivalTime              // CPU 空闲快进
    }
    const start = currentTime
    const end = start + p.burstTime
    gantt.push({ pid: p.id, start, end })
    currentTime = end

    const finishTime = end
    const turnaroundTime = finishTime - p.arrivalTime
    const weightedTurnaroundTime = turnaroundTime / p.burstTime
    results.push({
      id: p.id, finishTime, turnaroundTime,
      weightedTurnaroundTime: parseFloat(weightedTurnaroundTime.toFixed(2)),
    })
  }
  return { gantt, results }
}
```

### 2.2 RR（时间片轮转）

**算法思想**：维护就绪队列，每个进程轮流执行一个时间片；未完成则回队尾，新到达的进程动态入队。

**伪代码**：

```
算法 RR(processes, timeQuantum)
输入: processes, timeQuantum — 时间片长度
输出: { gantt[], results[] }

1.  queue ← processes 深拷贝，每项增加 remaining ← burstTime
2.  按 arrivalTime 升序排列 queue
3.  currentTime ← 0, readyQueue ← [], index ← 0
4.  gantt ← [], finishMap ← {}
5.
6.  function EnqueueArrived():
7.      while index < |queue| and queue[index].arrivalTime ≤ currentTime do
8.          readyQueue.push(queue[index])
9.          index ← index + 1
10.     end while
11. end function
12.
13. EnqueueArrived()
14.
15. while readyQueue 非空 do
16.     p ← readyQueue.shift()                    // 取队首
17.     execTime ← min(timeQuantum, p.remaining)
18.     gantt.push({ pid: p.id, start: currentTime, end: currentTime + execTime })
19.     currentTime ← currentTime + execTime
20.     p.remaining ← p.remaining - execTime
21.
22.     EnqueueArrived()                          // 新到达进程入队
23.
24.     if p.remaining > 0 then
25.         readyQueue.push(p)                    // 未完成 → 回队尾
26.     else
27.         finishMap[p.id] ← currentTime         // 完成
28.     end if
29.
30.     // CPU 空闲：就绪队列空但还有未到达进程
31.     if readyQueue 为空 and index < |queue| then
32.         currentTime ← queue[index].arrivalTime
33.         EnqueueArrived()
34.     end if
35. end while
36.
37. // 后处理：合并相邻同 pid 的甘特图段
38. mergedGantt ← 合并 gantt 中 pid 相同且首尾相连的相邻段
39.
40. for each p ∈ processes do
41.     计算 turnaroundTime = finishMap[p.id] - p.arrivalTime
42.     计算 weightedTurnaroundTime = turnaroundTime / p.burstTime
43. end for
44.
45. return { gantt: mergedGantt, results }
```

**关键代码 — 就绪队列循环**：

```js
export function rr(processes, timeQuantum) {
  const queue = processes.map((p) => ({ ...p, remaining: p.burstTime }))
  queue.sort((a, b) => a.arrivalTime - b.arrivalTime)

  const gantt = []
  const finishMap = {}
  let currentTime = 0
  const readyQueue = []
  let index = 0

  function enqueueArrived() {
    while (index < queue.length && queue[index].arrivalTime <= currentTime) {
      readyQueue.push(queue[index])
      index++
    }
  }

  enqueueArrived()

  while (readyQueue.length > 0) {
    const p = readyQueue.shift()
    const execTime = Math.min(timeQuantum, p.remaining)

    gantt.push({ pid: p.id, start: currentTime, end: currentTime + execTime })
    currentTime += execTime
    p.remaining -= execTime

    enqueueArrived()                           // 关键：推进时间后检查新到达的进程

    if (p.remaining > 0) {
      readyQueue.push(p)                       // 未完成，回队尾
    } else {
      finishMap[p.id] = currentTime
    }

    if (readyQueue.length === 0 && index < queue.length) {
      currentTime = queue[index].arrivalTime   // CPU 空闲快进
      enqueueArrived()
    }
  }
  // ... 合并甘特图段 + 计算结果
}
```

**关键代码 — 相邻同 pid 甘特图段合并**：

```js
// 合并连续的相同 pid 甘特图段（减少视觉碎片）
const mergedGantt = []
for (const seg of gantt) {
  const last = mergedGantt[mergedGantt.length - 1]
  if (last && last.pid === seg.pid && last.end === seg.start) {
    last.end = seg.end       // 延长上一个段
  } else {
    mergedGantt.push({ ...seg })
  }
}
```

### 2.3 SJF 抢占式（SRTF — 最短剩余时间优先）

**算法思想**：每个时间单位扫描所有已到达进程，选 `remaining` 最小的执行；新短进程到达可抢占当前进程。

**伪代码**：

```
算法 SJF_Preemptive(processes)
输入: processes
输出: { gantt[], results[] }

1.  procs ← processes 深拷贝，每项增加 remaining ← burstTime
2.  currentTime ← 0, completed ← 0, lastPid ← null
3.  gantt ← [], finishMap ← {}
4.
5.  while completed < |procs| do
6.      // 选已到达进程中 remaining 最小的
7.      shortest ← null
8.      for each p ∈ procs do
9.          if p.remaining > 0 and p.arrivalTime ≤ currentTime then
10.             if shortest = null or p.remaining < shortest.remaining then
11.                 shortest ← p
12.             end if
13.         end if
14.     end for
15.
16.     if shortest = null then
17.         // 无就绪进程 → 快进到下一个进程的到达时间
18.         currentTime ← min{ p.arrivalTime | p ∈ procs, p.remaining > 0 }
19.         continue
20.     end if
21.
22.     // 抢占检测：进程切换时记录甘特图段
23.     if lastPid ≠ shortest.id then
24.         if gantt 非空 then
25.             gantt[last].end ← currentTime    // 结束上一段
26.         end if
27.         gantt.push({ pid: shortest.id, start: currentTime, end: currentTime })
28.         lastPid ← shortest.id
29.     end if
30.
31.     // 每步只执行 1 个时间单位
32.     shortest.remaining ← shortest.remaining - 1
33.     currentTime ← currentTime + 1
34.
35.     if shortest.remaining = 0 then
36.         finishMap[shortest.id] ← currentTime
37.         completed ← completed + 1
38.         gantt[last].end ← currentTime
39.         lastPid ← null
40.     end if
41. end while
42.
43. validGantt ← 过滤掉 start < end 的空段
44. 计算各进程指标
45. return { gantt: validGantt, results }
```

**关键代码 — 逐时间单位抢占**：

```js
export function sjfPreemptive(processes) {
  const procs = processes.map((p) => ({ ...p, remaining: p.burstTime }))
  const gantt = []
  const finishMap = {}
  let currentTime = 0
  let completed = 0
  let lastPid = null

  while (completed < procs.length) {
    // 找出已到达且剩余时间最短的进程
    let shortest = null
    for (const p of procs) {
      if (p.remaining > 0 && p.arrivalTime <= currentTime) {
        if (!shortest || p.remaining < shortest.remaining) {
          shortest = p
        }
      }
    }

    if (!shortest) {
      // CPU 空闲，快进到下一个到达时间
      const nextArrival = procs
        .filter((p) => p.remaining > 0)
        .reduce((min, p) => Math.min(min, p.arrivalTime), Infinity)
      currentTime = nextArrival
      continue
    }

    // 抢占：进程切换时记录甘特图段边界
    if (lastPid !== shortest.id) {
      if (gantt.length > 0) {
        gantt[gantt.length - 1].end = currentTime   // 闭合上一段
      }
      gantt.push({ pid: shortest.id, start: currentTime, end: currentTime })
      lastPid = shortest.id
    }

    // 逐时间单位执行
    shortest.remaining--
    currentTime++

    if (shortest.remaining === 0) {
      finishMap[shortest.id] = currentTime
      completed++
      gantt[gantt.length - 1].end = currentTime
      lastPid = null
    }
  }
  // ...
}
```

### 2.4 HRN（最高响应比优先）

**算法思想**：非抢占式，每次选择响应比 `R = (等待时间 + 服务时间) / 服务时间` 最大的进程。

**伪代码**：

```
算法 HRN(processes)
输入: processes
输出: { gantt[], results[] }

1.  remaining ← processes 深拷贝
2.  currentTime ← 0
3.  gantt ← [], results ← []
4.
5.  while remaining 非空 do
6.      best ← null, bestRatio ← -1, bestIndex ← -1
7.
8.      for i ← 0 to |remaining|-1 do
9.          p ← remaining[i]
10.         if p.arrivalTime ≤ currentTime then
11.             waitTime ← currentTime - p.arrivalTime
12.             ratio ← (waitTime + p.burstTime) / p.burstTime     // 响应比公式
13.             if ratio > bestRatio then
14.                 bestRatio ← ratio
15.                 best ← p
16.                 bestIndex ← i
17.             end if
18.         end if
19.     end for
20.
21.     if best = null then
22.         currentTime ← min{ p.arrivalTime | p ∈ remaining }     // 快进
23.         continue
24.     end if
25.
26.     // 非抢占：一次性执行完
27.     start ← currentTime
28.     end ← start + best.burstTime
29.     gantt.push({ pid: best.id, start, end })
30.     currentTime ← end
31.
32.     计算 finishTime, turnaroundTime, weightedTurnaroundTime
33.     results.push({ ... })
34.     remaining.splice(bestIndex, 1)        // 移除已完成进程
35. end while
36.
37. return { gantt, results }
```

**关键代码 — 响应比计算**：

```js
export function hrn(processes) {
  const remaining = processes.map((p) => ({ ...p }))
  const gantt = []
  const results = []
  let currentTime = 0

  while (remaining.length > 0) {
    let best = null
    let bestRatio = -1
    let bestIndex = -1

    for (let i = 0; i < remaining.length; i++) {
      const p = remaining[i]
      if (p.arrivalTime <= currentTime) {
        const waitTime = currentTime - p.arrivalTime
        const ratio = (waitTime + p.burstTime) / p.burstTime   // R = (W+S)/S
        if (ratio > bestRatio) {
          bestRatio = ratio
          best = p
          bestIndex = i
        }
      }
    }

    if (!best) {
      const nextArrival = remaining.reduce(
        (min, p) => Math.min(min, p.arrivalTime), Infinity
      )
      currentTime = nextArrival
      continue
    }

    const start = currentTime
    const end = start + best.burstTime
    gantt.push({ pid: best.id, start, end })
    currentTime = end
    // ... 计算指标
    remaining.splice(bestIndex, 1)
  }
  return { gantt, results }
}
```

## 三、模块二：生产者-消费者同步

**源码文件**：`src/algorithms/producerConsumer.js`

**核心机制**：使用三个经典信号量（empty / full / mutex）+ 状态机模拟多线程并发。由于 JavaScript 为单线程，采用**随机调度器**在每步随机选择一个可行动线程推进。

---

### 3.1 信号量与数据结构

| 信号量 | 初值 | 含义 |
|--------|------|------|
| `empty` | bufferSize | 空闲缓冲位数 |
| `full` | 0 | 已占用缓冲位数 |
| `mutex` | 1 | 互斥访问（1=可用，0=被占用） |

| 线程状态 | 含义 |
|---------|------|
| `idle` | 空闲 |
| `waiting_empty` | 等待空闲缓冲位 P(empty) |
| `waiting_mutex` | 等待互斥锁 P(mutex) |
| `done` | 已完成一次生产/消费 |
| `waiting_full` | 等待产品 P(full) |
| `waiting_turn` | 等待继续取（连续取 n 个时） |

---

### 3.2 生产者状态机

```
         ┌──────────┐
         │   idle   │ ←──────────────────────┐
         └────┬─────┘                        │
              │ start                         │
              ▼                              │
    ┌─────────────────┐                      │
    │ waiting_empty    │                      │
    │  (等待 P(empty)) │                      │
    └────────┬────────┘                      │
             │ empty > 0 → empty--            │
             ▼                               │
    ┌─────────────────┐                      │
    │ waiting_mutex    │                      │
    │  (等待 P(mutex)) │                      │
    └────────┬────────┘                      │
             │ mutex > 0 → 进入临界区         │
             ▼                               │
    ┌─────────────────┐                      │
    │     done         │                      │
    │  放入产品到缓冲区 │                      │
    │  V(mutex) V(full)│──────────────────────┘
    └─────────────────┘
```

**伪代码**：

```
生产者状态转移:

case 'start':
    p.state ← 'waiting_empty'

case 'try_empty':                    // P(empty) 操作
    if empty > 0 then
        empty ← empty - 1
        p.state ← 'waiting_mutex'
    end if

case 'try_mutex':                    // P(mutex) + 临界区 + V 操作
    if mutex > 0 and mutexHolder = null then
        mutex ← mutex - 1
        mutexHolder ← { type: 'producer', id }
        // === 临界区 ===
        slot ← buffer 中第一个 null 位置
        totalProduced ← totalProduced + 1
        buffer[slot] ← totalProduced
        // ==============
        mutex ← mutex + 1            // V(mutex)
        mutexHolder ← null
        full ← full + 1              // V(full)
        p.state ← 'idle'
    end if
```

### 3.3 消费者状态机

```
         │   idle   │ ←──────────────────────────────┐
         └────┬─────┘                                │
              │ start                                 │
              ▼                                      │
    ┌─────────────────┐                              │

    │     done         │                │              │

             │ count < continuousN?    │              │
             │  是 → waiting_turn ─────┘              │
             │  否 → idle ───────────────────────────┘
             ▼
    ┌─────────────────┐

    └─────────────────┘
```

**伪代码**：

```
消费者状态转移:

case 'start':
    if consumerCount[id] > 0 then      // 正在连续取中
        c.state ← 'waiting_turn'
    else
        c.state ← 'waiting_full'
    end if

case 'try_full':                       // P(full) 操作
    if full > 0 then
        full ← full - 1
        c.state ← 'waiting_mutex'
    end if

case 'try_mutex':                      // P(mutex) + 临界区 + V 操作
    if mutex > 0 and mutexHolder = null then
        mutex ← mutex - 1
        mutexHolder ← { type: 'consumer', id }
        // === 临界区 ===
        slot ← buffer 中第一个非 null 位置
        buffer[slot] ← null
        consumerCount[id] ← consumerCount[id] + 1
        // ==============
        mutex ← mutex + 1              // V(mutex)
        mutexHolder ← null
        empty ← empty + 1              // V(empty)

        if consumerCount[id] ≥ continuousN then
            consumerCount[id] ← 0
            c.state ← 'idle'           // 取了 n 个，完成
        else
            c.state ← 'waiting_turn'   // 还需继续取
        end if
    end if

case 'try_turn':
    c.state ← 'waiting_full'
```

### 3.4 主调度循环

**伪代码**：

```
算法 Simulate(config)
输入: config { bufferSize, numProducers, numConsumers, continuousN, maxSteps }
输出: { states[], stats }

1.  初始化信号量: empty ← bufferSize, full ← 0, mutex ← 1
2.  初始化缓冲区: buffer[0..bufferSize-1] ← null
3.  初始化生产者和消费者的状态数组（全部 idle）
4.  初始化 consumerCount[] ← 全 0
5.  stepNum ← 0, totalProduced ← 0, totalConsumed ← 0
6.  Snapshot("初始状态")
7.
8.  while stepNum < maxSteps do
9.      stepNum ← stepNum + 1
10.     candidates ← []
11.
12.     // 收集生产者候选动作
13.     for each p ∈ producers do
14.         根据 p.state 加入对应候选（'start' | 'try_empty' | 'try_mutex'）
15.     end for
16.
17.     // 收集消费者候选动作
18.     for each c ∈ consumers do
19.         根据 c.state 加入对应候选
20.         ('start' | 'try_full' | 'try_mutex' | 'try_turn')
21.     end for
22.
23.     if candidates 为空 then
24.         记录 "所有线程阻塞"；break              // 死锁或全部完成
25.     end if
26.
27.     chosen ← RandomSelect(candidates)          // 随机选择一个候选执行
28.
29.     if chosen.type = 'producer' then
30.         执行生产者状态转移(chosen.action)
31.     else
32.         执行消费者状态转移(chosen.action)
33.     end if
34.
35.     if 有实际操作发生 then Snapshot()
36. end while
37.
38. return { states, stats: { totalProduced, totalConsumed } }
```

**关键代码 — 随机调度器核心**：

```js

          continuousN = 3, maxSteps = 50 } = config

  let empty = bufferSize, full = 0, mutex = 1
  const buffer = new Array(bufferSize).fill(null)
  const producers = /* ... */, consumers = /* ... */
  const consumerCount = new Array(numConsumers).fill(0)
  let mutexHolder = null, stepNum = 0, acted = false

  // ... 初始化状态 ...

  while (stepNum < maxSteps) {
    stepNum++
    const candidates = []

    // 生产者候选：根据当前 state 决定可执行的动作
    for (const p of producers) {
      if (p.state === 'idle')
        candidates.push({ type: 'producer', id: p.id, action: 'start' })
      else if (p.state === 'waiting_empty')
        candidates.push({ type: 'producer', id: p.id, action: 'try_empty' })
      else if (p.state === 'waiting_mutex')
        candidates.push({ type: 'producer', id: p.id, action: 'try_mutex' })
    }

    // 消费者候选
    for (const c of consumers) {
      if (c.state === 'idle')
        candidates.push({ type: 'consumer', id: c.id, action: 'start' })
      else if (c.state === 'waiting_full')
        candidates.push({ type: 'consumer', id: c.id, action: 'try_full' })
      else if (c.state === 'waiting_mutex')
        candidates.push({ type: 'consumer', id: c.id, action: 'try_mutex' })
      else if (c.state === 'waiting_turn')
        candidates.push({ type: 'consumer', id: c.id, action: 'try_turn' })
    }

    if (candidates.length === 0) break          // 全部阻塞或完成

    const chosen = candidates[Math.floor(Math.random() * candidates.length)]

    if (chosen.type === 'producer') {
      // ... 执行生产者状态转移 ...
    } else {
      // ... 执行消费者状态转移 ...
    }

    if (acted) snapshot(/* ... */)              // 记录状态快照
  }

  return { states, stats: { totalProduced, totalConsumed } }
}
```

### 4.1 安全性检查 — `checkSafety()`

**算法思想**：寻找安全序列——不断寻找 `Need[i] ≤ Work` 的进程，模拟其完成并释放资源，若所有进程都能完成则为安全状态。

**伪代码**：

```
算法 CheckSafety(max, allocation, available)
输入:
    max[n][m]        — n 个进程对 m 种资源的最大需求
    allocation[n][m] — 已分配矩阵
    available[m]     — 可用资源向量
输出: { safe, sequence[], need[][], steps[] }

1.  n ← |max|, m ← |available|
2.
3.  // 计算 Need 矩阵
4.  for i ← 0 to n-1 do
5.      for j ← 0 to m-1 do
6.          need[i][j] ← max[i][j] - allocation[i][j]
7.      end for
8.  end for
9.
10. work ← copy(available)
11. finish[0..n-1] ← false
12. sequence ← []
13.
14. repeat
15.     found ← false
16.     for i ← 0 to n-1 do
17.         if not finish[i] then
18.             // 检查 Need[i][*] ≤ Work[*]（对所有 j 满足）
19.             canAllocate ← true
20.             for j ← 0 to m-1 do
21.                 if need[i][j] > work[j] then
22.                     canAllocate ← false; break
23.                 end if
24.             end for
25.
26.             if canAllocate then
27.                 // 模拟进程 i 完成，释放其资源
28.                 for j ← 0 to m-1 do
29.                     work[j] ← work[j] + allocation[i][j]
30.                 end for
31.                 finish[i] ← true
32.                 sequence.push(i)
33.                 found ← true
34.                 break            // 重新从第一个进程扫描
35.             end if
36.         end if
37.     end for
38. until not found
39.
40. safe ← (finish 所有元素都为 true)
41. return { safe, sequence, need, steps }
```

**关键代码**：

```js
export function checkSafety(max, allocation, available) {
  const n = max.length
  const m = available.length

  // 计算 Need 矩阵
  const need = max.map((row, i) =>
    row.map((v, j) => v - allocation[i][j])
  )

  const work = [...available]
  const finish = new Array(n).fill(false)
  const sequence = []
  const steps = []

  let found = true
  while (found) {
    found = false
    for (let i = 0; i < n; i++) {
      if (!finish[i]) {
        // 检查 need[i] <= work（所有维度）
        const canAllocate = need[i].every((v, j) => v <= work[j])
        if (canAllocate) {
          // 模拟进程完成，释放资源
          for (let j = 0; j < m; j++) {
            work[j] += allocation[i][j]
          }
          finish[i] = true
          sequence.push(i)
          steps.push({
            process: i,
            workBefore: work.map((v, j) => v - allocation[i][j]),
            workAfter: [...work],
            finished: [...finish],
          })
          found = true
          break    // 重新扫描（释放的资源可能唤醒其他进程）
        }
      }
    }
  }

  const safe = finish.every((f) => f)
  return { safe, sequence: safe ? sequence : [], need, steps }
}
```

```
初始: Work = Available = [3, 3, 2]

第1轮: P1 Need=[1,2,2] ≤ [3,3,2] ✓ → P1 完成
       Work = [3,3,2] + [2,0,0] = [5,3,2]
第2轮: P3 Need=[0,0,2] ≤ [5,3,2] ✓ → P3 完成
       Work = [5,3,2] + [2,1,1] = [7,4,3]
第3轮: P0 Need=[7,4,3] ≤ [7,4,3] ✓ → P0 完成
       Work = [7,4,3] + [0,1,0] = [7,5,3]
第4轮: P2 Need=[6,0,0] ≤ [7,5,3] ✓ → P2 完成
       Work = [7,5,3] + [3,0,2] = [10,5,5]
第5轮: P4 Need=[4,3,1] ≤ [10,5,5] ✓ → P4 完成
```

### 4.2 资源请求 — `requestResources()`

**算法思想**：银行家算法的核心——即使资源足够，若分配后系统进入不安全状态也拒绝。

**伪代码**：

```
算法 RequestResources(max, allocation, available, pid, request)
输入:
    pid       — 请求进程索引
    request[m] — 请求的资源向量
输出: { granted, safe, message, sequence[] }

1.  计算 need[i][j] ← max[i][j] - allocation[i][j]（对所有 i, j）
2.
3.  // 第1步：Request ≤ Need?（不能超过声明的最大需求）
4.  for j ← 0 to m-1 do
5.      if request[j] > need[pid][j] then
6.          return { granted: false, message: "请求超过最大需求" }
7.      end if
8.  end for
9.
10. // 第2步：Request ≤ Available?（系统有足够资源）
11. for j ← 0 to m-1 do
12.     if request[j] > available[j] then
13.         return { granted: false, message: "资源不足" }
14.     end if
15. end for
16.
17. // 第3步：试探分配
18. newAllocation ← copy(allocation)
19. for j ← 0 to m-1 do
20.     newAllocation[pid][j] ← newAllocation[pid][j] + request[j]
21. end for
22. newAvailable[j] ← available[j] - request[j]
23.
24. // 第4步：对试探状态执行安全性检查
25. { safe, sequence } ← CheckSafety(max, newAllocation, newAvailable)
26.
27. if safe then
28.     return { granted: true, message: "可以分配", sequence }
29. else
30.     return { granted: false, message: "分配后不安全，不予分配" }
31. end if
```

**关键代码**：

```js
export function requestResources(max, allocation, available, pid, request) {
  const m = available.length

  // 计算 Need
  const need = max.map((row, i) =>
    row.map((v, j) => v - allocation[i][j])
  )

  // 1. Request ≤ Need?
  for (let j = 0; j < m; j++) {
    if (request[j] > need[pid][j]) {
      return {
        granted: false, safe: false,
        message: `请求超过最大需求！need[${pid}][${j}]=${need[pid][j]}，请求=${request[j]}`,
        sequence: [],
      }
    }
  }

  // 2. Request ≤ Available?
  for (let j = 0; j < m; j++) {
    if (request[j] > available[j]) {
      return {
        granted: false, safe: false,
        message: `资源不足！available[${j}]=${available[j]}，请求=${request[j]}`,
        sequence: [],
      }
    }
  }

  // 3. 试探分配
  const newAllocation = allocation.map((row, i) =>
    i === pid ? row.map((v, j) => v + request[j]) : [...row]
  )
  const newAvailable = available.map((v, j) => v - request[j])

  // 4. 安全性检查
  const { safe, sequence } = checkSafety(max, newAllocation, newAvailable)

  if (safe) {
    return { granted: true, safe: true, message: '可以分配，系统仍处于安全状态', sequence }
  } else {
    return { granted: false, safe: false, message: '分配后系统处于不安全状态，不予分配', sequence: [] }
  }
}
```

### 5.1 OPT（最优置换）

**算法思想**：淘汰**未来最远**才会被使用的页（理论最优，不可实现，仅作比较基准）。

**伪代码**：

```
算法 OPT(refSeq, frameCount)
输入: refSeq[0..L-1] — 页面引用串, frameCount — 物理块数
输出: { states[], totalFaults, faultRate }

1.  memory ← []              // 当前内存中的页
2.  totalFaults ← 0
3.
4.  for step ← 0 to L-1 do
5.      refPage ← refSeq[step]
6.      index ← memory 中查找 refPage 的位置
7.
8.      if index ≠ -1 then
9.          // 命中
10.         记录状态: fault ← false
11.     else
12.         // 缺页
13.         totalFaults ← totalFaults + 1
14.
15.         if |memory| < frameCount then
16.             memory.push(refPage)              // 有空闲块
17.         else
18.             // 淘汰：找未来最远才使用的页
19.             farthest ← -1
20.             replaceIndex ← 0
21.
22.             for i ← 0 to |memory|-1 do
23.                 page ← memory[i]
24.                 nextUse ← ∞                     // 默认永不使用
25.                 for k ← step+1 to L-1 do
26.                     if refSeq[k] = page then
27.                         nextUse ← k; break
28.                     end if
29.                 end for
30.
31.                 if nextUse > farthest then
32.                     farthest ← nextUse
33.                     replaceIndex ← i
34.                 end if
35.             end for
36.
37.             replacedPage ← memory[replaceIndex]
38.             memory[replaceIndex] ← refPage
39.         end if
40.
41.         记录状态: fault ← true, replacedPage
42.     end if
43. end for
44.
45. faultRate ← totalFaults / L × 100%
46. return { states, totalFaults, faultRate }
```

**关键代码 — 预测未来**：

```js
export function opt(refSeq, frameCount) {
  const states = []
  const memory = []
  let totalFaults = 0

  for (let step = 0; step < refSeq.length; step++) {
    const refPage = refSeq[step]
    const index = memory.indexOf(refPage)

    if (index !== -1) {
      // 命中
      states.push({ step, refPage, memory: [...memory], fault: false, replacedPage: null })
    } else {
      totalFaults++
      let replacedPage = null

      if (memory.length < frameCount) {
        memory.push(refPage)
      } else {
        // 找最晚使用或永不使用的页
        let farthest = -1
        let replaceIndex = 0

        for (let i = 0; i < memory.length; i++) {
          const page = memory[i]
          let nextUse = Infinity                     // 默认永不使用
          for (let j = step + 1; j < refSeq.length; j++) {
            if (refSeq[j] === page) {
              nextUse = j
              break
            }
          }
          if (nextUse > farthest) {
            farthest = nextUse
            replaceIndex = i
          }
        }

        replacedPage = memory[replaceIndex]
        memory[replaceIndex] = refPage
      }

      states.push({ step, refPage, memory: [...memory], fault: true, replacedPage })
    }
  }

  return {
    states, totalFaults,
    faultRate: parseFloat(((totalFaults / refSeq.length) * 100).toFixed(1)),
  }
}
```

### 5.2 LRU（最近最久未使用）

**算法思想**：淘汰**最久未被访问**的页。用数组模拟有序队列：队尾=最近使用，队首=最久未使用；命中时将页移到队尾。

**伪代码**：

```
算法 LRU(refSeq, frameCount)
输入: refSeq[0..L-1], frameCount
输出: { states[], totalFaults, faultRate }

// memory[] 语义：队首 = 最久未使用，队尾 = 最近使用

1.  memory ← []
2.  totalFaults ← 0
3.
4.  for step ← 0 to L-1 do
5.      refPage ← refSeq[step]
6.      index ← memory 中查找 refPage 的位置
7.
8.      if index ≠ -1 then
9.          // 命中 → 将该页移到队尾（"刷新"为最近使用）
10.         memory 中删除位置 index 的元素
11.         memory.push(refPage)
12.         记录状态: fault ← false
13.     else
14.         // 缺页
15.         totalFaults ← totalFaults + 1
16.         if |memory| < frameCount then
17.             memory.push(refPage)               // 有空闲块
18.         else
19.             replacedPage ← memory.shift()      // 淘汰队首（最久未使用）
20.             memory.push(refPage)               // 新页入队尾
21.         end if
22.         记录状态: fault ← true, replacedPage
23.     end if
24. end for
25.
26. faultRate ← totalFaults / L × 100%
27. return { states, totalFaults, faultRate }
```

**关键代码 — 命中时"刷新"位置**：

```js
export function lru(refSeq, frameCount) {
  const states = []
  const memory = []         // 队尾=最近使用，队首=最久未使用
  let totalFaults = 0

  for (let step = 0; step < refSeq.length; step++) {
    const refPage = refSeq[step]
    const index = memory.indexOf(refPage)

    if (index !== -1) {
      // 命中：移出原位置，推到队尾（标记为最近使用）
      memory.splice(index, 1)
      memory.push(refPage)
      states.push({ step, refPage, memory: [...memory], fault: false, replacedPage: null })
    } else {
      totalFaults++
      let replacedPage = null

      if (memory.length < frameCount) {
        memory.push(refPage)
      } else {
        replacedPage = memory.shift()    // 弹出队首（最久未使用）
        memory.push(refPage)             // 新页入队尾
      }

      states.push({ step, refPage, memory: [...memory], fault: true, replacedPage })
    }
  }

  return { states, totalFaults,
    faultRate: parseFloat(((totalFaults / refSeq.length) * 100).toFixed(1)) }
}
```

### 6.1 SSTF（最短寻道时间优先）

**算法思想**：贪心策略，每次选择距离当前磁头位置**最近**的请求服务。

**伪代码**：

```
算法 SSTF(requests, startPos)
输入: requests[] — 磁道请求序列, startPos — 初始磁头位置
输出: { order[], totalDistance, path[] }

1.  remaining ← copy(requests)
2.  currentPos ← startPos
3.  totalDistance ← 0
4.  order ← []
5.  path ← [{ step: 0, pos: startPos }]
6.
7.  while remaining 非空 do
8.      // 找到距离当前磁头最近的请求
9.      minDist ← ∞
10.     minIndex ← -1
11.
12.     for i ← 0 to |remaining|-1 do
13.         dist ← |remaining[i] - currentPos|
14.         if dist < minDist then
15.             minDist ← dist
16.             minIndex ← i
17.         end if
18.     end for
19.
20.     target ← remaining[minIndex]
21.     totalDistance ← totalDistance + |target - currentPos|
22.     currentPos ← target
23.     order.push(target)
24.     path.push({ step: ++stepCount, pos: target, target })
25.     remaining 中删除 minIndex 位置的元素
26. end while
27.
28. return { order, totalDistance, path }
```

**关键代码**：

```js
export function sstf(requests, startPos) {
  const remaining = [...requests]
  const order = []
  const path = [{ step: 0, pos: startPos, target: null }]
  let currentPos = startPos
  let totalDistance = 0
  let stepCount = 0

  while (remaining.length > 0) {
    let minDist = Infinity
    let minIndex = -1

    for (let i = 0; i < remaining.length; i++) {
      const dist = Math.abs(remaining[i] - currentPos)
      if (dist < minDist) {
        minDist = dist
        minIndex = i
      }
    }

    const target = remaining[minIndex]
    totalDistance += Math.abs(target - currentPos)
    currentPos = target
    order.push(target)
    stepCount++
    path.push({ step: stepCount, pos: currentPos, target })
    remaining.splice(minIndex, 1)
  }

  return { order, totalDistance, path }
}
```

### 6.2 SCAN（电梯算法）

**算法思想**：磁头像电梯一样沿一个方向移动到最远的请求，然后折返服务另一侧的请求。

**伪代码**：

```
算法 SCAN(requests, startPos, direction)
输入: requests[], startPos, direction ∈ {'up', 'down'}
输出: { order[], totalDistance, path[] }

1.  sorted ← requests 升序排列
2.  currentPos ← startPos
3.  totalDistance ← 0
4.
5.  // 分离两组请求
6.  greater ← sorted 中所有 ≥ currentPos 的元素（升序）
7.  less    ← sorted 中所有 < currentPos 的元素（升序）
8.
9.  if direction = 'up' then
10.     // 先向外（增大方向），到头后折返向内
11.     serveOrder ← concat(greater, reverse(less))
12. else
13.     // 先向内（减小方向），到头后折返向外
14.     serveOrder ← concat(reverse(less), greater)
15. end if
16.
17. order ← []
18. for each target ∈ serveOrder do
19.     totalDistance ← totalDistance + |target - currentPos|
20.     currentPos ← target
21.     order.push(target)
22.     path.push({ step: ++stepCount, pos: target, target })
23. end for
24.
25. return { order, totalDistance, path }
```

**关键代码 — 方向控制与分组**：

```js
export function scan(requests, startPos, direction = 'up') {
  const sorted = [...requests].sort((a, b) => a - b)
  const order = []
  const path = [{ step: 0, pos: startPos, target: null }]
  let currentPos = startPos
  let totalDistance = 0
  let stepCount = 0

  // 分离 >= 和 < 当前磁道的请求
  const greater = sorted.filter((t) => t >= currentPos)
  const less = sorted.filter((t) => t < currentPos)

  let serveOrder
  if (direction === 'up') {
    // 先向外（增大）服务，再折返向内
    serveOrder = [...greater, ...less.reverse()]
  } else {
    // 先向内（减小）服务，再折返向外
    serveOrder = [...less.reverse(), ...greater]
  }

  for (const target of serveOrder) {
    totalDistance += Math.abs(target - currentPos)
    currentPos = target
    order.push(target)
    stepCount++
    path.push({ step: stepCount, pos: currentPos, target })
  }

  return { order, totalDistance, path }
}
```

**SCAN 算法执行示例**：

```

请求: [98, 183, 37, 122, 14, 124, 65, 67]

greater (≥53): [65, 67, 98, 122, 124, 183]
less (<53):    [14, 37]
serveOrder:    [65, 67, 98, 122, 124, 183, 37, 14]
                                  ↑折返点

移动路径: 53 → 65 → 67 → 98 → 122 → 124 → 183 → 37 → 14
总寻道距离 = 236
```

核心 `renderItem` 函数动态计算每个矩形的坐标：

```js
function makeRectRender(dataArray, styleFn) {
  return (params, api) => {
    const item = dataArray[params.dataIndex]
    const yIndex = getYIndex(item.pid)
    const start = api.coord([item.value[0], yIndex])
    const end = api.coord([item.value[1], yIndex])
    const barHeight = api.size([0, 1])[1] * 0.55

    return styleFn({
      x: start[0],
      y: start[1] - barHeight / 2,
      w: Math.max(end[0] - start[0], 2),
      h: barHeight,
      item, api,
    })
  }
}
```

## 八、架构设计要点

### 8.1 算法与可视化分离

每个算法模块导出的是**纯函数**，不依赖任何 DOM/Vue/ECharts API：

```
算法函数(参数) → { gantt/results } 或 { states[] }
视图组件读取结果 → 渲染图表
```

这使得算法可独立测试（甚至可在 Node.js 环境运行），也方便增加新算法或替换可视化库。

### 8.2 状态快照机制

支持逐帧回放的关键设计：算法运行过程中在关键步骤调用 `snapshot()` 记录完整状态，返回 `states[]` 后由播放控件按帧索引渲染，用户可前进/后退/拖拽观察任意时刻的系统状态。

### 8.3 单线程并发模拟

由于 JavaScript 无法真正多线程，生产者-消费者模块采用**随机调度器 + 状态机**模拟并发：每步从所有可执行动作中随机选一个推进，等价于线程调度的随机性。
